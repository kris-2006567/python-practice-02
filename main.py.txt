#ПРАКТИЧЕСКАЯ РАБОТА 3. ВАРИАНТ 2. КОНВЕРТЕР (АРГУМЕНТЫ КОМАНДНОЙ СТРОКИ)

#Импортируем библиотеку argparse
# argparse — это специальная библиотека, которая помогает программе понимать аргументы, 
# которые пользователь вводит в командной строке (например, --list или --conversion).
import argparse

#Импортируем библиотеку datetime
# datetime — библиотека для работы с датой и временем.
# Она нужна нам, чтобы записывать в лог файл точное время, когда произошла ошибка.
import datetime

#Создаем функцию log_error для записи ошибок в файл
# def — ключевое слово для создания функции.
# log_error — имя функции (можно назвать как угодно).
# (error, message) — параметры функции: 
#   error — сама ошибка (ее тип и текст),
#   message — наше пояснение (при какой операции произошла ошибка).
def log_error(error, message):
    # with open(...) — открывает файл и автоматически закрывает его, когда код внутри закончится.
    # "errors.log" — имя файла, куда будем писать.
    # "a" (append) — режим добавления. Если файл есть, новые записи добавятся в конец, старые не сотрутся.
    # encoding="utf-8" — кодировка, чтобы русские буквы нормально сохранялись и не превращались в кракозябры.
    with open("errors.log", "a", encoding="utf-8") as f:
        # now — переменная для текущего времени.
        # datetime.datetime.now() — получает текущую дату и время с компьютера.
        # .strftime("%Y-%m-%d %H:%M:%S") — превращает время в красивый текст.
        # %Y — год, %m — месяц, %d — день, %H — часы, %M — минуты, %S — секунды.
        now = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        
        # f.write(...) — записывает текст в файл.
        # f"..." — f-строка, позволяет вставлять переменные прямо в текст.
        # [{now}] — вставляет текущее время в квадратные скобки.
        # {type(error).__name__} — определяет тип ошибки (например, ValueError или ZeroDivisionError).
        # \n — символ переноса на новую строку.
        f.write(f"[{now}] {type(error).__name__}:\n")
        # str(error) — превращает саму ошибку в текст и записывает ее.
        f.write(f"{str(error)}\n")
        # Дописываем наше пояснение, что мы делали, когда произошла ошибка.
        f.write(f"Message: {message}\n\n")

#Создаем парсер (обработчик) аргументов командной строки
# ArgumentParser — это как переводчик, который разбирает то, что написал пользователь в консоли.
# description — описание программы. Оно покажется, если пользователь введет --help.
parser = argparse.ArgumentParser(description="Конвертер единиц измерения")

#Добавляем позиционный аргумент (то есть без дефисов, вводится просто как число)
# parser.add_argument — добавляет новую команду в наш "переводчик".
# "value" — имя аргумента. По нему мы потом будем к нему обращаться (args.value).
# type=str — говорит, что мы ожидаем текст (строку). Число мы превратим потом.
# nargs="?" — (очень важно!) Означает, что аргумент НЕОБЯЗАТЕЛЬНЫЙ. 
# Если его не написать, программа не вылетит с ошибкой (это нужно для работы команды --list).
# help — подсказка, которая выведется при --help.
parser.add_argument("value", type=str, nargs="?", help="Значение для конвертации")

#Добавляем именованный аргумент для выбора типа перевода
# Именованные аргументы начинаются с двух дефисов --.
# "--conversion" — именно так пользователь будет это писать.
# type=str — ожидаем текст (например, celsius-fahrenheit).
parser.add_argument("--conversion", type=str, help="Тип конвертации (например, celsius-fahrenheit)")

#Добавляем аргумент для настройки точности (сколько знаков после запятой оставить)
# "--precision" — как пользователь это вызовет.
# type=int — ожидаем целое число (например, 2 или 4).
# default=2 — если пользователь не укажет точность, по умолчанию будет 2 знака.
parser.add_argument("--precision", type=int, default=2, help="Количество знаков после запятой")

#Добавляем флаг --list (показать список доступных переводов)
# action="store_true" — это особый тип аргумента-переключателя. 
# Если пользователь напишет --list, значение станет True (правда). Если не напишет — False (ложь).
parser.add_argument("--list", action="store_true", help="Показать список доступных конвертаций")

#Запускаем парсер и сохраняем результат в переменную args
# parse_args() — читает то, что ввел пользователь, и раскладывает по переменным.
# Теперь мы можем обращаться к ним так: args.value, args.conversion и т.д.
args = parser.parse_args()

#Начинаем блок try (попытаться)
# В этот блок мы кладем код, который МОЖЕТ вызвать ошибку (например, если ввели букву вместо числа).
# Если ошибка произойдет, программа резко перепрыгнет в блок except (внизу).
try:
    #Если пользователь ввел флаг --list (он равен True)
    if args.list:
        #Выводим список доступных конвертаций
        print("Доступные конвертации:")
        print("1. celsius-fahrenheit (Цельсий -> Фаренгейт)")
        print("2. km-miles (Километры -> Мили)")
        print("3. kg-pounds (Килограммы -> Фунты)")
        print("4. m-feet (Метры -> Футы)")
        print("5. liters-gallons (Литры -> Галлоны)")
        # exit() — полностью завершает программу. 
        # Нам не нужно ничего конвертировать, мы просто показали список и вышли.
        exit() 

    #Проверяем, ввел ли пользователь значение и тип конвертации.
    # args.value is None — проверяет, пустое ли значение.
    # or — логическое ИЛИ (если хотя бы одно пустое).
    if args.value is None or args.conversion is None:
        # raise ValueError(...) — принудительно вызывает ошибку с нашим текстом.
        # Мы делаем это сами, чтобы программа записала ошибку в лог и выдала сообщение.
        raise ValueError("Не указано значение или тип конвертации.")

    #Пытаемся превратить введенный текст (args.value) в дробное число (float).
    # Например, "100" превратится в 100.0. 
    # Если пользователь ввел "привет", тут случится ошибка ValueError.
    val = float(args.value)

    #Начинаем проверять, какой тип конвертации выбрал пользователь.
    # args.conversion хранит текст, который ввели после --conversion.
    if args.conversion == "celsius-fahrenheit":
        #Формула перевода Цельсия в Фаренгейт.
        result = val * 9 / 5 + 32
        #Сохраняем название операции для вывода.
        text = "Цельсий -> Фаренгейт"
    elif args.conversion == "km-miles":
        #Формула перевода километров в мили.
        result = val * 0.621371
        text = "Километры -> Мили"
    elif args.conversion == "kg-pounds":
        #Формула перевода килограммов в фунты.
        result = val * 2.20462
        text = "Килограммы -> Фунты"
    elif args.conversion == "m-feet":
        #Формула перевода метров в футы.
        result = val * 3.28084
        text = "Метры -> Футы"
    elif args.conversion == "liters-gallons":
        #Формула перевода литров в галлоны.
        result = val * 0.264172
        text = "Литры -> Галлоны"
    else:
        #Если пользователь ввел тип конвертации, которого нет в списке.
        # Вызываем ошибку с текстом, куда вставляем то, что он ввел.
        raise ValueError(f"Неизвестный тип конвертации: {args.conversion}")

    #Выводим результат на экран.
    print(f"Input value: {args.value}")
    print(f"Conversion: {text}")
    # round(result, args.precision) — округляет результат.
    # args.precision — число знаков после запятой (по умолчанию 2).
    print(f"Result: {round(result, args.precision)}")

#Если в блоке try произошла ошибка ValueError (неверные данные)
# as e — сохраняет информацию об ошибке в переменную e.
except ValueError as e:
    #Передаем ошибку и текст в нашу функцию, чтобы она записала это в файл errors.log.
    log_error(e, "Ошибка при выполнении конвертации")
    #Выводим понятное сообщение для пользователя.
    print(f"Ошибка: {e}")
    print("Проверьте введенные данные. Подробности записаны в errors.log")

#Если произошла ЛЮБАЯ другая ошибка, которую мы не предусмотрели.
except Exception as e:
    #Записываем ее в файл с пометкой "Непредвиденная ошибка".
    log_error(e, "Непредвиденная ошибка")
    #Выводим сообщение пользователю.
    print(f"Произошла непредвиденная ошибка: {e}")
