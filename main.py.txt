#Добавлен цикл для повторных преобразований и новые конвертеры

#Бесконечный цикл для повторения преобразований
while True:
    #Вывод меню конвертеров
    print("\nВыберите конвертер:")
    print("1. Градус Цельсия -> Градус Фаренгейта")
    print("2. Километры -> Мили")
    print("3. Килограммы -> Фунты")
    print("4. Метры -> Футы")
    print("5. Литры -> Галлоны")        # Новый конвертер
    print("6. Мегабайты -> Гигабайты")  # Новый конвертер

    #Блок try для обработки некорректного ввода
    try:
        #Получение номера конвертера
        vibor = int(input("Ваш выбор: "))
        
        #Обработка выбора
        if vibor == 1:
            c = float(input("Введите температуру: "))
            f = c * 9 / 5 + 32
            print(f"{c} C = {f:.1f} F")
        elif vibor == 2:
            km = float(input("Введите километры: "))
            mili = km * 0.621371
            print(f"{km} км = {mili:.2f} миль")
        elif vibor == 3:
            kg = float(input("Введите килограммы: "))
            funty = kg * 2.20462
            print(f"{kg} кг = {funty:.2f} фунтов")
        elif vibor == 4:
            m = float(input("Введите метры: "))
            futy = m * 3.28084
            print(f"{m} м = {futy:.2f} футов")
        elif vibor == 5:
            l = float(input("Введите литры: "))
            gallony = l * 0.264172
            print(f"{l} л = {gallony:.2f} галлонов")
        elif vibor == 6:
            mb = float(input("Введите мегабайты: "))
            gb = mb / 1024
            print(f"{mb} МБ = {gb:.2f} ГБ")
        else:
            print("Ошибка: неизвестный номер преобразования")

    #Если ввели текст вместо числа
    except ValueError:
        print("Ошибка: нужно ввести число")

    #Цикл для проверки ответа на продолжение
    while True:
        otvet = input("\nВыполнить ещё одно преобразование? (да/нет): ")
        if otvet.lower() == "да":
            break
        elif otvet.lower() == "нет":
            print("Работа программы завершена.")
            exit()
        else:
            print("Ошибка: введите 'да' или 'нет'")
